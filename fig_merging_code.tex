CSR csr_add(const CSR& A,
            const CSR& B) {
  for (int i = 0; i < m; ++i) {
    int p = A.rowptr[i];
    int q = B.rowptr[i];
    int pe = A.rowptr[i + 1];
    int qe = B.rowptr[i + 1];

    while (p < pe || q < qe) {
      int ca = (p < pe) ? A.col[p] : INF;
      int cb = (q < qe) ? B.col[q] : INF;
      if (ca < cb) {
        emit(ca, A.val[p]);
        ++p;
      } else if (cb < ca) {
        emit(cb, B.val[q]);
        ++q;
      } else {
        emit(ca, A.val[p] + B.val[q]);
        ++p; ++q;
      }
    }
    close_row();
  }
}
\end{lstlisting}

CSR csr_mul(const CSR& A,
            const CSR& B) {
  for (int i = 0; i < m; ++i) {
    int p = A.rowptr[i];
    int q = B.rowptr[i];
    int pe = A.rowptr[i + 1];
    int qe = B.rowptr[i + 1];

    while (p < pe && q < qe) {
      int ca = A.col[p];
      int cb = B.col[q];
      if (ca < cb) {
        ++p;
      } else if (cb < ca) {
        ++q;
      } else {
        emit(ca, A.val[p] * B.val[q]);
        ++p; ++q;
      }
    }
    close_row();
  }
}
\end{lstlisting}
